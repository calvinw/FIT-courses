Table Schemas to Give to AI

Paste this first to Gemini as you ask it to write queries for you. It shows the three tables, their columns, and how they connect.

CREATE TABLE channels (
  channel_id int NOT NULL,
  channel_name varchar(50) NOT NULL,
  PRIMARY KEY (channel_id)
);

CREATE TABLE users (
  user_id int NOT NULL,
  signup_date date,
  name varchar(100),
  email varchar(100),
  address varchar(255),
  PRIMARY KEY (user_id)
);

-- opened_email, clicked_link, and converted are 1 for yes and 0 for no.
-- session_duration is in seconds. NULL means no value was recorded.
CREATE TABLE interactions (
  interaction_id int NOT NULL,
  user_id int,
  channel_id int,
  campaign varchar(100),
  opened_email tinyint,
  clicked_link tinyint,
  session_duration int,
  pages_visited int,
  converted tinyint,
  PRIMARY KEY (interaction_id),
  FOREIGN KEY (user_id) REFERENCES users (user_id),
  FOREIGN KEY (channel_id) REFERENCES channels (channel_id)
);
